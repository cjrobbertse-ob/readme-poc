export const CEGMoney = (value, currency = "GBP") => {
  const cents = CEGCents(value);
  return cents === null ? "Enter an amount" : currency + " " + (cents / 100).toLocaleString("en-GB", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
};

export const CEGCents = (value) => {
  const text = String(value);
  if (!/^\d{1,7}(.\d{1,2})?$/.test(text)) return null;
  const [whole, fraction = ""] = text.split(".");
  return Number(whole) * 100 + Number(fraction.padEnd(2, "0"));
};

export const CEGValidate = (step, values, journey) => {
  if (step.kind === "bank" && !values.bank) return "Choose a bank to continue.";
  if (step.kind === "payer" && (!values.bank || (!journey.bankSelect && !values.account))) return "Choose your bank" + (journey.bankSelect ? "." : " and a saved paying account.");
  if (step.kind === "accounts" && !(journey.type === "ais" ? values.accounts.length : values.account)) return "Select at least one eligible account.";
  if (step.kind === "payment") {
    if (journey.file && !values.fileLoaded) return "Load the fictional payment file to continue.";
    const cents = CEGCents(values.amount);
    if (cents === null || cents <= 0) return "Enter an amount greater than zero, with at most two decimal places.";
    if (values.reference.length > 18) return "An optional payment reference must not exceed 18 characters.";
    if ((journey.scheduled || journey.standing) && (!CEGIsDate(values.date) || values.date <= CEGMeta.sampleDate)) return "Choose a valid date after 9 November 2026, the example's current date.";
    if (journey.standing && (!/^\d{1,3}$/.test(values.count) || Number(values.count) < 2 || Number(values.count) > 120)) return "Choose between 2 and 120 payments.";
    if (journey.standing && [values.firstAmount, values.finalAmount].some(v => CEGCents(v) === null || CEGCents(v) <= 0)) return "First and final payment amounts must be positive and have at most two decimal places.";
  }
  if (step.kind === "vrpRules") {
    const per = CEGCents(values.maxPayment);
    const period = CEGCents(values.maxMonth);
    if (per === null || period === null || per <= 0 || period < per) return "Set a positive per-payment maximum and a monthly maximum at least as large.";
    if (!values.ongoing && (!CEGIsDate(values.expiry) || values.expiry <= CEGMeta.sampleDate)) return "Choose a valid consent expiry after 9 November 2026.";
  }
  if (step.kind === "identifier" && ["modelA", "modelB"].includes(step.channel) && !/^[A-Z0-9-]{4,20}$/i.test(values.identifier)) return "Enter an example identifier of 4 to 20 letters, digits or hyphens; do not enter credentials.";
  if (step.kind === "cofPayer" && (!/^\d{8}$/.test(values.payerNumber) || !/^\d{2}-?\d{2}-?\d{2}$/.test(values.payerSort))) return "Enter an 8-digit example account number and 6-digit sort code.";
  if (step.kind === "cofPayer" && (values.payerNumber !== "00001042" || values.payerSort.replaceAll("-", "") !== "123456")) return "Only the fictional account 00001042 / 12-34-56 is supported. Use the example funding account; do not enter real account details.";
  if (step.kind === "serviceSetup" && !values.acceptedTerms) return "Accept the fictional service terms to continue.";
  if (step.kind === "vrpExecute" && (CEGCents(values.amount) === null || CEGCents(values.amount) <= 0)) return "Enter a positive payment amount with at most two decimal places.";
  return "";
};

export const CEGCheckVRP = (values, usedCents = 12000, revoked = false) => {
  if (revoked) return "The consent has been revoked. No payment was initiated.";
  const amount = CEGCents(values.amount);
  const per = CEGCents(values.maxPayment);
  const month = CEGCents(values.maxMonth);
  if (amount === null || per === null || month === null || amount <= 0) return "Invalid payment or consent parameters. No payment was initiated.";
  if (amount > per) return "The payment exceeds the per-payment maximum. No payment was initiated.";
  if (usedCents + amount > month) return "The payment exceeds the remaining calendar-month limit. No payment was initiated.";
  return "";
};

export const CEGDate = (date) => new Date(date + "T12:00:00Z").toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric", timeZone: "UTC" });

export const CEGIsDate = (value) => {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) return false;
  const date = new Date(value + "T12:00:00Z");
  return !Number.isNaN(date.getTime()) && date.toISOString().slice(0, 10) === value;
};

export const CEGSchedule = (v) => {
  const last = new Date(v.date + "T12:00:00Z");
  const periods = Number(v.count) - 1;
  if (v.frequency === "Weekly") last.setUTCDate(last.getUTCDate() + periods * 7);
  else {
    const day = last.getUTCDate();
    last.setUTCDate(1);
    last.setUTCMonth(last.getUTCMonth() + periods * (v.frequency === "Quarterly" ? 3 : 1));
    const lastDay = new Date(Date.UTC(last.getUTCFullYear(), last.getUTCMonth() + 1, 0)).getUTCDate();
    last.setUTCDate(Math.min(day, lastDay));
  }
  return Number.isNaN(last.getTime()) ? "Choose a valid schedule" : CEGDate(last.toISOString().slice(0, 10));
};

export const CEGContrastText = (hex) => {
  const rgb = hex.slice(1).match(/.{2}/g).map(x => parseInt(x, 16) / 255).map(x => x <= .04045 ? x / 12.92 : ((x + .055) / 1.055) ** 2.4);
  const luminance = rgb[0] * .2126 + rgb[1] * .7152 + rgb[2] * .0722;
  return (luminance + .05) / .05 > 1.05 / (luminance + .05) ? "#000000" : "#ffffff";
};
