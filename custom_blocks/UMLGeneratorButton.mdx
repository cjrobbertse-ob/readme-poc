function umlClamp(value, min, max) {
  return Math.min(max, Math.max(min, value));
}

function umlEscapeXml(value) {
  return String(value == null ? '' : value)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&apos;');
}

function umlDecodeHtmlEntities(value) {
  if (typeof document === 'undefined') return value;
  var textarea = document.createElement('textarea');
  textarea.innerHTML = value;
  return textarea.value;
}

function umlTruncate(value, maxLength) {
  var text = String(value == null ? '' : value);
  if (text.length <= maxLength) return text;
  return text.slice(0, Math.max(0, maxLength - 1)) + '…';
}

function umlUnique(values) {
  var seen = {};
  return values.filter(function (value) {
    if (seen[value]) return false;
    seen[value] = true;
    return true;
  });
}

function umlThemeFromMode(mode, props) {

}

function umlThemeModeFromText(value) {
  var text = String(value || '').toLowerCase();
  if (/(^|[\s_-])light($|[\s_-])/.test(text)) return 'light';
  if (/(^|[\s_-])dark($|[\s_-])/.test(text) || text.indexOf('night') !== -1) return 'dark';
  return '';
}

function umlThemeModeFromElementAttributes(element) {
  if (!element || !element.getAttribute) return '';

  var explicitAttributes = ['data-theme', 'data-color-mode', 'data-color-scheme', 'data-readme-color-scheme', 'color-scheme'];
  for (var index = 0; index < explicitAttributes.length; index += 1) {
    var mode = umlThemeModeFromText(element.getAttribute(explicitAttributes[index]));
    if (mode) return mode;
  }

  return umlThemeModeFromText(element.style && element.style.colorScheme);
}

function umlThemeModeFromElementClasses(element) {
  if (!element || !element.className) return '';
  return umlThemeModeFromText(String(element.className));
}

function umlParseRgbColor(value) {
  var match = String(value || '').match(/rgba?\((\d+),\s*(\d+),\s*(\d+)(?:,\s*([0-9.]+))?\)/i);
  if (!match || match[4] === '0') return null;

  return {
    red: Number(match[1]),
    green: Number(match[2]),
    blue: Number(match[3]),
  };
}

function umlRelativeLuminance(color) {
  if (!color) return null;

  function channel(value) {
    var normalized = value / 255;
    return normalized <= 0.03928 ? normalized / 12.92 : Math.pow((normalized + 0.055) / 1.055, 2.4);
  }

  return 0.2126 * channel(color.red) + 0.7152 * channel(color.green) + 0.0722 * channel(color.blue);
}

function umlThemeModeFromDocumentPaint(doc) {
  var docWindow = doc && doc.defaultView ? doc.defaultView : typeof window !== 'undefined' ? window : null;
  if (!doc || !docWindow || !docWindow.getComputedStyle) return '';

  var elements = [doc.body, doc.documentElement].filter(Boolean);
  for (var index = 0; index < elements.length; index += 1) {
    try {
      var styles = docWindow.getComputedStyle(elements[index]);
      var background = umlParseRgbColor(styles && styles.backgroundColor);
      var luminance = umlRelativeLuminance(background);
      if (luminance !== null) return luminance < 0.5 ? 'dark' : 'light';
    } catch (error) {
      // Ignore inaccessible computed styles and continue to the next document.
    }
  }

  return '';
}

function umlDetectReadMeThemeMode(props) {
  var requested = props && props.theme ? String(props.theme).toLowerCase() : 'auto';
  if (requested === 'light' || requested === 'dark') return requested;

  var documents = umlAvailableDocuments();

  for (var index = 0; index < documents.length; index += 1) {
    var doc = documents[index];
    var elements = [doc.documentElement, doc.body].filter(Boolean);
    for (var attrIndex = 0; attrIndex < elements.length; attrIndex += 1) {
      var attrMode = umlThemeModeFromElementAttributes(elements[attrIndex]);
      if (attrMode) return attrMode;
    }
  }

  for (var paintIndex = 0; paintIndex < documents.length; paintIndex += 1) {
    var paintMode = umlThemeModeFromDocumentPaint(documents[paintIndex]);
    if (paintMode) return paintMode;
  }

  for (var classDocIndex = 0; classDocIndex < documents.length; classDocIndex += 1) {
    var classDoc = documents[classDocIndex];
    var classElements = [classDoc.documentElement, classDoc.body].filter(Boolean);
    for (var classIndex = 0; classIndex < classElements.length; classIndex += 1) {
      var classMode = umlThemeModeFromElementClasses(classElements[classIndex]);
      if (classMode) return classMode;
    }
  }

  if (typeof window !== 'undefined' && window.matchMedia) {
    try {
      if (window.matchMedia('(prefers-color-scheme: dark)').matches) return 'dark';
    } catch (error) {
      // Fall back to the light brand theme when media queries are unavailable.
    }
  }

  return 'light';
}

function umlObjectKeys(object) {

  });
}

function umlResolveApiFilename(api) {
  var text = String(api || '').trim();
  if (!text) throw new Error('Add an api prop, for example api="PIS".');

  if (UML_API_ALIASES[text]) return UML_API_ALIASES[text];

  var normalized = text
    .replace(/\.(json|ya?ml)$/i, '')
    .replace(/[_\s]+/g, '-');
  var aliasKey = Object.keys(UML_API_ALIASES).find(function (key) {
    return key.toLowerCase() === text.toLowerCase();
  });

  return aliasKey ? UML_API_ALIASES[aliasKey] : normalized;
}

function umlFindApiDefinitions(value, results, depth) {
  if (!value || depth > 10) return;

  if (Array.isArray(value)) {
    value.forEach(function (item) {
      umlFindApiDefinitions(item, results, depth + 1);
    });
    return;
  }

  if (typeof value !== 'object') return;

  if (Array.isArray(value.apiDefinitions)) {
    value.apiDefinitions.forEach(function (definition) {
      if (definition && definition.source && definition.source.sync_url) {
        results.push(definition);
      }
    });
  }

  Object.keys(value).forEach(function (key) {
    if (key === 'apiDefinitions') return;
    umlFindApiDefinitions(value[key], results, depth + 1);
  });
}

function umlAvailableDocuments() {
  if (typeof document === 'undefined') return [];

  var docs = [document];
  if (typeof window !== 'undefined') {
    try {
      if (window.parent && window.parent.document) docs.push(window.parent.document);
    } catch (error) {
      // Parent document is cross-origin on some ReadMe surfaces.
    }

    try {
      if (window.top && window.top.document) docs.push(window.top.document);
    } catch (error) {
      // Top document is cross-origin on some ReadMe surfaces.
    }
  }

  return docs.filter(function (doc, index) {
    return doc && docs.indexOf(doc) === index;
  });
}

function umlReadScriptJson(script) {
  if (!script) return '';

  var candidates = [
    script.textContent,
    script.innerHTML,
    script.getAttribute && script.getAttribute('data-json'),
    script.getAttribute && script.getAttribute('data-props'),
  ];

  for (var index = 0; index < candidates.length; index += 1) {
    if (candidates[index]) return candidates[index];
  }

  return '';
}

function umlParseReadMeState(raw) {
  try {
    return JSON.parse(raw);
  } catch (firstError) {
    try {
      return JSON.parse(umlDecodeHtmlEntities(raw));
    } catch (secondError) {
      throw new Error('ReadMe page state could not be parsed as JSON.');
    }
  }
}

function umlFindDefinitionsInScript(script) {
  var raw = umlReadScriptJson(script);
  if (!raw || raw.indexOf('apiDefinitions') === -1 || raw.indexOf('sync_url') === -1) return [];

  var definitions = [];
  umlFindApiDefinitions(umlParseReadMeState(raw), definitions, 0);
  return definitions;
}

function umlEscapeRegExp(value) {
  return String(value || '').replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

function umlNormalizeHtmlForOpenApiSearch(value) {
  var text = umlDecodeHtmlEntities(String(value || ''));
  text = text.replace(/\\u002F/gi, '/').replace(/\\\//g, '/');

  try {
    text = decodeURIComponent(text);
  } catch (error) {
    // Ignore malformed percent-encoding in unrelated page content.
  }

  return text;
}

function umlAbsoluteUrl(url, baseUrl) {
  try {
    return new URL(url, baseUrl || window.location.origin).href;
  } catch (error) {
    return url;
  }
}

function umlFindDefinitionsInDocument(doc) {
  if (!doc || !doc.querySelectorAll) return [];

  var scripts = Array.prototype.slice.call(
    doc.querySelectorAll('script#ssr-props, script[id="ssr-props"], script[type="application/json"], script')
  );
  var definitions = [];

  scripts.some(function (script) {
    try {
      definitions = umlFindDefinitionsInScript(script);
      return definitions.length > 0;
    } catch (error) {
      return false;
    }
  });

  return definitions;
}

function umlParseHtmlDocument(html) {
  if (!html || typeof DOMParser === 'undefined') return [];
  return new DOMParser().parseFromString(html, 'text/html');
}

function umlFindDefinitionsInHtml(html) {
  var parsedDocument = umlParseHtmlDocument(html);
  return parsedDocument ? umlFindDefinitionsInDocument(parsedDocument) : [];
}

function umlFindOpenApiUrlInHtml(html, api, baseUrl) {
  var wantedFilename = umlResolveApiFilename(api);
  var escapedFilename = umlEscapeRegExp(wantedFilename);
  var text = umlNormalizeHtmlForOpenApiSearch(html);
  var patterns = [
    new RegExp('(https?:\\/\\/[^"\\\'<>\\s)]+\\/' + escapedFilename + '\\.(?:json|ya?ml)(?:[?#][^"\\\'<>\\s)]*)?)', 'i'),
    new RegExp('(\\/(?:branches|versions)\\/[^"\\\'<>\\s)]+\\/apis\\/' + escapedFilename + '\\.(?:json|ya?ml)(?:[?#][^"\\\'<>\\s)]*)?)', 'i'),
    new RegExp(
      '(?:sync_url|url|href|uri)["\\\']?\\s*:\\s*["\\\']([^"\\\']*\\/' +
        escapedFilename +
        '\\.(?:json|ya?ml)(?:[?#][^"\\\']*)?)["\\\']',
      'i'
    ),
  ];

  for (var index = 0; index < patterns.length; index += 1) {
    var match = text.match(patterns[index]);
    if (match) return umlAbsoluteUrl(match[1] || match[0], baseUrl);
  }

  return '';
}

function umlFindReadMeVersionInHtml(html) {
  var text = umlNormalizeHtmlForOpenApiSearch(html);
  var patterns = [
    /"urlManagerOpts"[\s\S]{0,800}?"version"\s*:\s*"([^"]+)"/i,
    /"version"\s*:\s*"([^"]+)"/i,
    /readme-version["'][^>]*content=["']([^"']+)["']/i,
    /\/(?:branches|versions)\/([^/"'<>?#\s)]+)\/apis\//i,
    /\/(?:branches|versions)\/([^/"'<>?#\s)]+)/i,
  ];

  for (var index = 0; index < patterns.length; index += 1) {
    var match = text.match(patterns[index]);
    if (match && match[1]) return match[1];
  }

  return '';
}

function umlCandidatePageUrls() {
  if (typeof window === 'undefined') return [];

  var urls = [];

  function addUrl(value) {
    if (typeof value === 'string' && value && urls.indexOf(value) === -1 && !/^about:|^blob:|^data:/i.test(value)) {
      urls.push(value);
    }
  }

  try {
    addUrl(window.location && window.location.href);
  } catch (error) {
    // Location may be unavailable in some embedded previews.
  }

  try {
    addUrl(document && document.referrer);
  } catch (error) {
    // Referrer may be unavailable.
  }

  try {
    var canonical = document && document.querySelector && document.querySelector('link[rel="canonical"]');
    addUrl(canonical && canonical.href);
  } catch (error) {
    // Canonical link may be unavailable.
  }

  try {
    addUrl(window.parent && window.parent.location && window.parent.location.href);
  } catch (error) {
    // Parent location can be cross-origin in the ReadMe editor.
  }

  try {
    addUrl(window.top && window.top.location && window.top.location.href);
  } catch (error) {
    // Top location can be cross-origin in the ReadMe editor.
  }

  return umlUnique(
    urls.map(function (url) {
      return String(url).split('#')[0];
    })
  );
}

async function umlFetchPageHtml(url) {
  var response = await fetch(url, { credentials: 'include' });
  if (!response.ok) return '';
  return await response.text();
}

async function umlFetchPageDefinitions() {
  var urls = umlCandidatePageUrls();

  for (var index = 0; index < urls.length; index += 1) {
    try {
      var response = await fetch(urls[index], { credentials: 'include' });
      if (!response.ok) continue;

      var definitions = umlFindDefinitionsInHtml(await response.text());
      if (definitions.length) return definitions;
    } catch (error) {
      // Try the next candidate URL; the final error explains that discovery failed.
    }
  }

  return [];
}

async function discoverReadMeOpenApiUrlFromPageHtml(api) {
  if (typeof document === 'undefined') {
    throw new Error('ReadMe page state is unavailable outside the browser.');
  }

  var documents = umlAvailableDocuments();

  for (var docIndex = 0; docIndex < documents.length; docIndex += 1) {
    try {
      var doc = documents[docIndex];
      var html = doc.documentElement && doc.documentElement.outerHTML;
      var found = umlFindOpenApiUrlInHtml(html, api, doc.location && doc.location.href);
      if (found) return found;
    } catch (error) {
      // Ignore inaccessible parent/top documents.
    }
  }

  var urls = umlCandidatePageUrls();
  for (var urlIndex = 0; urlIndex < urls.length; urlIndex += 1) {
    try {
      var fetchedHtml = await umlFetchPageHtml(urls[urlIndex]);
      var fetchedFound = umlFindOpenApiUrlInHtml(fetchedHtml, api, urls[urlIndex]);
      if (fetchedFound) return fetchedFound;
    } catch (error) {
      // Ignore fetch failures and try the next candidate.
    }
  }

  throw new Error('Could not infer the OpenAPI URL from the current page HTML for api="' + api + '".');
}

async function discoverReadMeOpenApiSyncUrl(api) {
  var wantedFilename = umlResolveApiFilename(api);
  var documents = umlAvailableDocuments();
  var definitions = [];

  documents.some(function (doc) {
    definitions = umlFindDefinitionsInDocument(doc);
    return definitions.length > 0;
  });

  if (!definitions.length) definitions = await umlFetchPageDefinitions();

  if (!definitions.length) {
    throw new Error('Could not find ReadMe page state in this frame, its parent, or the current page HTML.');
  }

  var match = definitions.find(function (definition) {
    var url = definition && definition.source && definition.source.sync_url;
    return (
      typeof url === 'string' &&
      new RegExp('/' + wantedFilename.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '\\.ya?ml(?:[?#].*)?$', 'i').test(url)
    );
  });

  if (!match) {
    throw new Error('Could not find a ReadMe OpenAPI sync_url for "' + api + '" (' + wantedFilename + ').');
  }

  return match.source.sync_url;
}

function deriveOpenApiJsonUrl(syncUrl) {
  var url = String(syncUrl || '');
  if (!url) throw new Error('The ReadMe OpenAPI sync_url was empty.');
  if (/\.json(?:[?#].*)?$/i.test(url)) return url;
  if (!/\.ya?ml(?:[?#].*)?$/i.test(url)) {
    throw new Error('The ReadMe OpenAPI sync_url did not point to a YAML or JSON file: ' + url);
  }
  return url.replace(/\.ya?ml(?=([?#].*)?$)/i, '.json');
}

function umlExplicitOpenApiUrl(options) {
  if (!options) return '';
  return options.openApiUrl || options.sourceUrl || options.url || '';
}

function umlGithubRefsForReadMeVersion(version) {
  var raw = String(version || '').trim();
  if (!raw) return [];

  var cleaned = raw.replace(/^refs\/(?:tags|heads)\//i, '').replace(/^\/+|\/+$/g, '');
  var withoutV = cleaned.replace(/^v/i, '');
  var refs = [];

  function addRef(ref) {
    if (ref && refs.indexOf(ref) === -1) refs.push(ref);
  }

  addRef('refs/tags/' + cleaned);
  addRef('refs/tags/v' + withoutV);

  var updateMatch = withoutV.match(/^(\d+\.\d+)(?:\.0)?[-_]?update[-_]?(\d+)$/i);
  if (updateMatch) addRef('refs/tags/v' + updateMatch[1] + '-Update-' + updateMatch[2]);

  var leadingVersion = withoutV.match(/^(\d+\.\d+(?:\.\d+)?)/);
  if (leadingVersion) addRef('refs/tags/v' + leadingVersion[1]);

  addRef('refs/heads/' + cleaned);
  addRef('refs/heads/' + withoutV);
  addRef('refs/heads/v' + withoutV);

  return refs;
}

function umlFindReadMeVersionInUrl(url) {
  var match = String(url || '').match(/\/(?:branches|versions)\/([^/?#]+)/i);
  return match && match[1] ? decodeURIComponent(match[1]) : '';
}

async function umlInferReadMeVersion() {
  var documents = umlAvailableDocuments();

  for (var docIndex = 0; docIndex < documents.length; docIndex += 1) {
    try {
      var doc = documents[docIndex];
      var foundInDocument = umlFindReadMeVersionInHtml(doc.documentElement && doc.documentElement.outerHTML);
      if (foundInDocument) return foundInDocument;
      var foundInLocation = umlFindReadMeVersionInUrl(doc.location && doc.location.href);
      if (foundInLocation) return foundInLocation;
    } catch (error) {
      // Ignore inaccessible parent/top documents.
    }
  }

  var urls = umlCandidatePageUrls();
  for (var urlIndex = 0; urlIndex < urls.length; urlIndex += 1) {
    var foundInUrl = umlFindReadMeVersionInUrl(urls[urlIndex]);
    if (foundInUrl) return foundInUrl;

    try {
      var html = await umlFetchPageHtml(urls[urlIndex]);
      var foundInHtml = umlFindReadMeVersionInHtml(html);
      if (foundInHtml) return foundInHtml;
    } catch (error) {
      // Ignore fetch failures and try the next candidate.
    }
  }

  return '';
}

function umlBuildOpenBankingRawJsonUrls(api, readMeVersion) {
  var filename = umlResolveApiFilename(api);
  return umlGithubRefsForReadMeVersion(readMeVersion).map(function (githubRef) {
    return (
      'https://raw.githubusercontent.com/OpenBankingUK/read-write-api-specs/' +
      githubRef.replace(/^\/+|\/+$/g, '') +
      '/dist/openapi/' +
      filename +
      '.json'
    );
  });
}

function umlResolvePointer(documentRoot, pointer) {
  if (!pointer || pointer.charAt(0) !== '#') {
    throw new Error('Unsupported external OpenAPI reference: ' + pointer);
  }

  var parts = pointer
    .slice(2)
    .split('/')
    .map(function (part) {
      return part.replace(/~1/g, '/').replace(/~0/g, '~');
    });

  return parts.reduce(function (current, part) {
    if (!current || typeof current !== 'object' || !(part in current)) {
      throw new Error('Could not resolve OpenAPI reference: ' + pointer);
    }
    return current[part];
  }, documentRoot);
}

function umlMergeSchemas(schemas) {
  return schemas.reduce(function (merged, schema) {
    var next = schema || {};
    var mergedRequired = umlUnique((merged.required || []).concat(next.required || []));
    var mergedProperties = Object.assign({}, merged.properties || {}, next.properties || {});

    return Object.assign({}, merged, next, {
      description: merged.description || next.description,
      required: mergedRequired,
      properties: mergedProperties,
      allOf: undefined,
    });
  }, {});
}

function umlNormalizeSchema(schema, documentRoot, refStack) {
  if (!schema || typeof schema !== 'object') return {};

  var stack = refStack || [];
  var working = schema;

  if (working.$ref) {
    if (stack.indexOf(working.$ref) !== -1) {
      return Object.assign({}, working, { circularRef: true });
    }

    var resolved = umlNormalizeSchema(umlResolvePointer(documentRoot, working.$ref), documentRoot, stack.concat(working.$ref));
    var siblings = Object.assign({}, working);
    delete siblings.$ref;
    working = Object.assign({}, resolved, siblings);
  }

  if (Array.isArray(working.allOf)) {
    var mergedAllOf = umlMergeSchemas(
      working.allOf.map(function (item) {
        return umlNormalizeSchema(item, documentRoot, stack);
      })
    );
    var local = Object.assign({}, working);
    delete local.allOf;
    working = umlMergeSchemas([mergedAllOf, local]);
  }

  return working;
}

function umlSchemaType(schema, documentRoot) {
  var normalized = umlNormalizeSchema(schema, documentRoot);
  if (normalized.oneOf) return 'oneOf';
  if (normalized.anyOf) return 'anyOf';
  if (normalized.type) return normalized.type;
  if (normalized.properties) return 'object';
  if (normalized.items) return 'array';
  if (normalized.enum) return 'string';
  return 'any';
}

function umlChoiceTooltip(schema, documentRoot) {
  var normalized = umlNormalizeSchema(schema, documentRoot);
  var choices = normalized.oneOf || normalized.anyOf;
  if (!choices) return '';

  var label = normalized.oneOf ? 'One of' : 'Any of';
  return (
    label +
    ':\n' +
    choices
      .map(function (choice, index) {
        var resolved = umlNormalizeSchema(choice, documentRoot);
        var type = umlSchemaType(resolved, documentRoot);
        var title = resolved.title || resolved.description || resolved.enum;
        return ' - Option ' + (index + 1) + ': ' + type + (title ? ' (' + String(title).slice(0, 140) + ')' : '');
      })
      .join('\n')
  );
}

function umlFieldTooltip(schema, documentRoot) {
  var normalized = umlNormalizeSchema(schema, documentRoot);
  var lines = [];
  var type = umlSchemaType(normalized, documentRoot);

  if (type === 'array') {
    type = 'array<' + umlSchemaType(normalized.items || {}, documentRoot) + '>';
  }

  lines.push('Type: ' + type);
  ['format', 'pattern', 'minimum', 'maximum', 'minLength', 'maxLength', 'minItems', 'maxItems'].forEach(function (key) {
    if (normalized[key] !== undefined) lines.push(key + ': ' + normalized[key]);
  });

  if (Array.isArray(normalized.enum)) {
    lines.push('Values:');
    normalized.enum.forEach(function (value) {
      lines.push(' - ' + value);
    });
  }

  if (normalized.description) lines.push('Description: ' + normalized.description);

  var choiceText = umlChoiceTooltip(normalized, documentRoot);
  if (choiceText) lines.push(choiceText);

  return lines.join('\n');
}

function umlCardinality(schema, isRequired, documentRoot) {
  var normalized = umlNormalizeSchema(schema, documentRoot);
  var isArray = umlSchemaType(normalized, documentRoot) === 'array';
  var min = isRequired ? 1 : 0;
  var max = isRequired ? 1 : 1;

  if (isArray) {
    min = normalized.minItems !== undefined ? normalized.minItems : min;
    max = normalized.maxItems !== undefined ? normalized.maxItems : '*';
  }

  return '[' + min + '..' + max + ']';
}

function umlIsObjectLike(schema, documentRoot) {
  var normalized = umlNormalizeSchema(schema, documentRoot);
  var type = umlSchemaType(normalized, documentRoot);
  return type === 'object' && !!normalized.properties && !normalized.oneOf && !normalized.anyOf;
}

function umlChildSchema(schema, documentRoot) {
  var normalized = umlNormalizeSchema(schema, documentRoot);
  if (umlSchemaType(normalized, documentRoot) === 'array') {
    var itemSchema = umlNormalizeSchema(normalized.items || {}, documentRoot);
    return umlIsObjectLike(itemSchema, documentRoot) ? itemSchema : null;
  }
  return umlIsObjectLike(normalized, documentRoot) ? normalized : null;
}

function umlRefKey(schema) {
  if (!schema || typeof schema !== 'object') return '';
  if (schema.$ref) return schema.$ref;
  if (schema.items && schema.items.$ref) return schema.items.$ref;
  return '';
}

function umlParseSelector(selector) {
  var text = String(selector || '').trim();
  if (!text) throw new Error('Add a schema prop, for example schema="OBWriteDomesticConsent4.properties.Data".');

  var alias = '';
  var path = text;
  var aliasIndex = text.lastIndexOf(':');
  if (aliasIndex > -1) {
    path = text.slice(0, aliasIndex);
    alias = text.slice(aliasIndex + 1).trim();
  }

  var parts = path.split('.').filter(Boolean);
  if (!parts.length) throw new Error('Schema selector was empty.');

  return {
    parts: parts,
    alias: alias || parts[parts.length - 1],
  };
}

function umlTryParseSelector(selector) {
  try {
    return umlParseSelector(selector);
  } catch (error) {
    return null;
  }
}

function umlConfigurationWarnings(api, schema) {
  var warnings = [];
  if (!String(api || '').trim()) warnings.push('Add an api prop, for example api="PIS".');
  if (!String(schema || '').trim()) {
    warnings.push('Add a schema prop, for example schema="OBWriteDomesticConsent4".');
  }
  return warnings;
}

function resolveOpenApiSchemaSelector(openApiDocument, selector) {
  var parsed = umlParseSelector(selector);
  var schemas =
    openApiDocument &&
    openApiDocument.components &&
    openApiDocument.components.schemas;

  if (!schemas || !schemas[parsed.parts[0]]) {
    throw new Error('Could not find top-level OpenAPI schema "' + parsed.parts[0] + '".');
  }

  var current = schemas[parsed.parts[0]];

  for (var index = 1; index < parsed.parts.length; index += 1) {
    var segment = parsed.parts[index];
    var normalized = umlNormalizeSchema(current, openApiDocument);

    if (segment === 'properties') {
      index += 1;
      var propertyName = parsed.parts[index];
      if (!propertyName) throw new Error('Schema selector ended after ".properties".');
      if (!normalized.properties || !normalized.properties[propertyName]) {
        throw new Error('Could not find property "' + propertyName + '" in selector "' + selector + '".');
      }
      current = normalized.properties[propertyName];
      continue;
    }

    if (segment === 'items') {
      if (!normalized.items) throw new Error('Could not find items in selector "' + selector + '".');
      current = normalized.items;
      continue;
    }

    if (normalized.properties && normalized.properties[segment]) {
      current = normalized.properties[segment];
      continue;
    }

    if (normalized[segment] !== undefined) {
      current = normalized[segment];
      continue;
    }

    throw new Error('Could not resolve selector segment "' + segment + '" in "' + selector + '".');
  }

  return {
    title: parsed.alias,
    schema: umlNormalizeSchema(current, openApiDocument),
  };
}

function umlBuildNode(openApiDocument, schema, title, refPath, ancestors, ancestorRefs) {
  var normalized = umlNormalizeSchema(schema, openApiDocument);
  var required = normalized.required || [];
  var fields = [];
  var children = [];
  var currentAncestors = ancestors || [];
  var currentAncestorRefs = ancestorRefs || [];

  umlObjectKeys(normalized.properties).forEach(function (propertyName) {
    var propertySchema = normalized.properties[propertyName];
    var propertyNormalized = umlNormalizeSchema(propertySchema, openApiDocument);
    var cardinality = umlCardinality(propertyNormalized, required.indexOf(propertyName) !== -1, openApiDocument);
    var childSchema = umlChildSchema(propertyNormalized, openApiDocument);
    var nextPath = refPath + '.properties.' + propertyName;
    var refKey = umlRefKey(propertySchema);

    if (
      childSchema &&
      currentAncestors.indexOf(nextPath) === -1 &&
      (!refKey || currentAncestorRefs.indexOf(refKey) === -1)
    ) {
      children.push({
        name: propertyName,
        cardinality: cardinality,
        tooltip: umlFieldTooltip(propertyNormalized, openApiDocument),
        node: umlBuildNode(
          openApiDocument,
          childSchema,
          propertyName,
          nextPath,
          currentAncestors.concat(nextPath),
          refKey ? currentAncestorRefs.concat(refKey) : currentAncestorRefs
        ),
      });
      return;
    }

    fields.push({
      name: propertyName,
      cardinality: cardinality,
      tooltip: umlFieldTooltip(propertyNormalized, openApiDocument),
    });
  });

  return {
    id: 'uml-node-' + refPath.replace(/[^a-z0-9]+/gi, '-'),
    title: title,
    tooltip: normalized.description || title,
    fields: fields,
    children: children,
    width: UML_LAYOUT.boxMinWidth,
    height: UML_LAYOUT.headerHeight + Math.max(1, fields.length) * UML_LAYOUT.rowHeight + UML_LAYOUT.boxPaddingX,
    x: 0,
    y: 0,
    subtreeHeight: 0,
  };
}

function umlMeasureNode(node) {
  var longest = node.title.length;
  node.fields.forEach(function (field) {
    longest = Math.max(longest, (field.name + ' ' + field.cardinality).length);
  });
  node.width = umlClamp(longest * 7.2 + UML_LAYOUT.boxPaddingX * 2, UML_LAYOUT.boxMinWidth, UML_LAYOUT.boxMaxWidth);
  node.height = UML_LAYOUT.headerHeight + Math.max(1, node.fields.length) * UML_LAYOUT.rowHeight + UML_LAYOUT.boxPaddingX;
  node.children.forEach(function (child) {
    umlMeasureNode(child.node);
  });
}

function umlLayoutNode(node, columnLefts, depth, top) {
  node.x = columnLefts[depth];
  node.y = top;

  var childTop = top;
  node.children.forEach(function (child, index) {
    if (index > 0) childTop += UML_LAYOUT.boxGapY;
    umlLayoutNode(child.node, columnLefts, depth + 1, childTop);
    childTop += child.node.subtreeHeight;
  });
}

function umlCollectColumnWidths(node, depth, widths) {
  widths[depth] = Math.max(widths[depth] || 0, node.width);
  node.children.forEach(function (child) {
    umlCollectColumnWidths(child.node, depth + 1, widths);
  });
}

function umlColumnLefts(widths) {
  var lefts = [];
  var nextLeft = UML_LAYOUT.margin;

  widths.forEach(function (width, index) {
    lefts[index] = nextLeft;
    nextLeft += width + UML_LAYOUT.columnGap;
  });

  return lefts;
}

function umlCalculateSubtreeHeight(node) {
  node.children.forEach(function (child) {
    umlCalculateSubtreeHeight(child.node);
  });

  var childrenHeight = node.children.reduce(function (total, child, index) {
    return total + child.node.subtreeHeight + (index === 0 ? 0 : UML_LAYOUT.boxGapY);
  }, 0);
  node.subtreeHeight = Math.max(node.height, childrenHeight);
}

function umlWalkNodes(node, callback) {
  callback(node);
  node.children.forEach(function (child) {
    umlWalkNodes(child.node, callback);
  });
}

function umlRenderNode(node) {
  var parts = [];
  var titleY = node.y + 22;
  var rowStartY = node.y + UML_LAYOUT.headerHeight + 20;

  parts.push(
    '<g class="uml-node" id="' +
      umlEscapeXml(node.id) +
      '">' +
      '<title>' +
      umlEscapeXml(node.tooltip) +
      '</title>' +
      '<rect class="uml-box" x="' +
      node.x +
      '" y="' +
      node.y +
      '" width="' +
      node.width +
      '" height="' +
      node.height +
      '" rx="8"/>' +
      '<rect class="uml-header" x="' +
      node.x +
      '" y="' +
      node.y +
      '" width="' +
      node.width +
      '" height="' +
      UML_LAYOUT.headerHeight +
      '" rx="8"/>' +
      '<path class="uml-header-seam" d="M ' +
      node.x +
      ' ' +
      (node.y + UML_LAYOUT.headerHeight) +
      ' H ' +
      (node.x + node.width) +
      '"/>' +
      '<text class="uml-title" x="' +
      (node.x + node.width / 2) +
      '" y="' +
      titleY +
      '" text-anchor="middle">' +
      umlEscapeXml(umlTruncate(node.title, 54)) +
      '</text>'
  );

  if (!node.fields.length) {
    parts.push(
      '<text class="uml-field uml-empty" x="' +
        (node.x + UML_LAYOUT.boxPaddingX) +
        '" y="' +
        rowStartY +
        '">No scalar fields</text>'
    );
  } else {
    node.fields.forEach(function (field, index) {
      parts.push(
        '<text class="uml-field" x="' +
          (node.x + UML_LAYOUT.boxPaddingX) +
          '" y="' +
          (rowStartY + index * UML_LAYOUT.rowHeight) +
          '">' +
          '<title>' +
          umlEscapeXml(field.tooltip) +
          '</title>' +
          umlEscapeXml(umlTruncate(field.name + ' ' + field.cardinality, 64)) +
          '</text>'
      );
    });
  }

  parts.push('</g>');
  return parts.join('');
}

function umlRenderEdge(parent, child) {
  var fromX = parent.x + parent.width;
  var fromY = parent.y + UML_LAYOUT.headerHeight / 2;
  var toX = child.node.x;
  var toY = child.node.y + UML_LAYOUT.headerHeight / 2;
  var busX = fromX + Math.min(44, Math.max(24, (toX - fromX) / 2));
  var labelText = child.cardinality.replace(/^\[|\]$/g, '');
  var labelX = busX + Math.max(34, (toX - busX) / 2);
  var labelY = toY - 8;

  return (
    '<g class="uml-edge">' +
    '<title>' +
    umlEscapeXml(child.tooltip) +
    '</title>' +
    '<path d="M ' +
    fromX +
    ' ' +
    fromY +
    ' H ' +
    busX +
    ' V ' +
    toY +
    ' H ' +
    toX +
    '" marker-start="url(#umlDiamond)" marker-end="url(#umlArrow)"/>' +
    '<rect class="uml-edge-label-bg" x="' +
    (labelX - 24) +
    '" y="' +
    (labelY - 14) +
    '" width="48" height="18" rx="9"/>' +
    '<text class="uml-edge-label" x="' +
    labelX +
    '" y="' +
    labelY +
    '" text-anchor="middle">' +
    umlEscapeXml(labelText) +
    '</text>' +
    '</g>'
  );
}

function generateUmlSvg(openApiDocument, selector, theme) {
  var umlTheme = theme || UML_THEMES.light;
  var selected = resolveOpenApiSchemaSelector(openApiDocument, selector);
  var rootNode = umlBuildNode(openApiDocument, selected.schema, selected.title, selected.title, [selected.title], []);

  umlMeasureNode(rootNode);
  umlCalculateSubtreeHeight(rootNode);
  var columnWidths = [];
  umlCollectColumnWidths(rootNode, 0, columnWidths);
  umlLayoutNode(rootNode, umlColumnLefts(columnWidths), 0, UML_LAYOUT.margin);

  var maxX = 0;
  var maxY = 0;
  umlWalkNodes(rootNode, function (node) {
    maxX = Math.max(maxX, node.x + node.width);
    maxY = Math.max(maxY, node.y + node.height);
  });

  var width = Math.ceil(maxX + UML_LAYOUT.margin);
  var height = Math.ceil(maxY + UML_LAYOUT.margin);
  var edges = [];
  var nodes = [];

  umlWalkNodes(rootNode, function (node) {
    node.children.forEach(function (child) {
      edges.push(umlRenderEdge(node, child));
    });
    nodes.push(umlRenderNode(node));
  });

  return (
    '<?xml version="1.0" encoding="UTF-8"?>' +
    '<svg xmlns="http://www.w3.org/2000/svg" width="' +
    width +
    '" height="' +
    height +
    '" viewBox="0 0 ' +
    width +
    ' ' +
    height +
    '" role="img" aria-label="' +
    umlEscapeXml(selected.title) +
    ' schema diagram">' +
    '<title>' +
    umlEscapeXml(selected.title) +
    ' schema diagram</title>' +
    '<defs>' +
    '<marker id="umlArrow" viewBox="0 0 10 10" refX="9" refY="5" markerWidth="9" markerHeight="9" orient="auto-start-reverse"><path class="uml-marker" d="M 0 0 L 10 5 L 0 10 z"/></marker>' +
    '<marker id="umlDiamond" viewBox="0 0 16 10" refX="0" refY="5" markerWidth="16" markerHeight="10" orient="auto"><path class="uml-marker" d="M 0 5 L 8 10 L 16 5 L 8 0 z"/></marker>' +
    '<style><![CDATA[' +
    'svg{background:' +
    umlTheme.background +
    ';font-family:Metropolis,Inter,Arial,sans-serif;color:' +
    umlTheme.text +
    '} .uml-box{fill:' +
    umlTheme.childFill +
    ';stroke:' +
    umlTheme.primaryBorder +
    ';stroke-width:2;filter:drop-shadow(0 4px 10px ' +
    umlTheme.shadow +
    ')}.uml-header{fill:' +
    umlTheme.primary +
    ';stroke:' +
    umlTheme.primaryBorder +
    ';stroke-width:2}.uml-header-seam{stroke:' +
    umlTheme.primaryBorder +
    ';stroke-width:2}.uml-title{fill:' +
    umlTheme.primaryText +
    ';font-weight:700;font-size:' +
    UML_LAYOUT.titleFontSize +
    'px}.uml-field{fill:' +
    umlTheme.field +
    ';font-size:' +
    UML_LAYOUT.fontSize +
    'px}.uml-empty{fill:' +
    umlTheme.muted +
    ';font-style:italic}.uml-edge path{fill:none;stroke:' +
    umlTheme.primaryBorder +
    ';stroke-width:2}.uml-marker{fill:' +
    (umlTheme.marker || umlTheme.accent || umlTheme.primaryBorder) +
    '}.uml-edge-label-bg{fill:' +
    umlTheme.accentFill +
    ';stroke:' +
    umlTheme.accent +
    ';stroke-width:1}.uml-edge-label{fill:' +
    umlTheme.text +
    ';font-size:12px;font-weight:600}' +
    ']]></style>' +
    '</defs>' +
    edges.join('') +
    nodes.join('') +
    '</svg>'
  );
}

function umlSvgToDataUrl(svg) {
  return 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svg || '');
}

function umlDownloadFilename(schema) {
  var parsed = umlParseSelector(schema);
  return parsed.alias.replace(/[^a-z0-9._-]+/gi, '-').replace(/^-+|-+$/g, '') || 'schema-diagram';
}

async function umlFetchOpenApiDocument(api, options) {
  var explicitUrl = umlExplicitOpenApiUrl(options);
  var candidateUrls = [];
  var discoveryErrors = [];

  if (explicitUrl) {
    candidateUrls.push(deriveOpenApiJsonUrl(explicitUrl));
  } else {
    try {
      candidateUrls.push(deriveOpenApiJsonUrl(await discoverReadMeOpenApiUrlFromPageHtml(api)));
    } catch (htmlDiscoveryError) {
      discoveryErrors.push(htmlDiscoveryError && htmlDiscoveryError.message);
      try {
        candidateUrls.push(deriveOpenApiJsonUrl(await discoverReadMeOpenApiSyncUrl(api)));
      } catch (stateDiscoveryError) {
        discoveryErrors.push(stateDiscoveryError && stateDiscoveryError.message);
      }
    }

    try {
      var readMeVersion = await umlInferReadMeVersion();
      candidateUrls = candidateUrls.concat(umlBuildOpenBankingRawJsonUrls(api, readMeVersion));
    } catch (versionError) {
      discoveryErrors.push(versionError && versionError.message);
    }
  }

  candidateUrls = umlUnique(candidateUrls.filter(Boolean));
  if (!candidateUrls.length) {
    throw new Error(
      'Could not infer an OpenAPI URL for api="' +
        api +
        '". ' +
        discoveryErrors.filter(Boolean).join(' ')
    );
  }

  var fetchErrors = [];

  for (var index = 0; index < candidateUrls.length; index += 1) {
    var jsonUrl = candidateUrls[index];
    try {
      var isRawGithubUrl = /^https:\/\/raw\.githubusercontent\.com\//i.test(jsonUrl);
      var response = await fetch(jsonUrl, { credentials: isRawGithubUrl ? 'omit' : 'same-origin' });

      if (!response.ok) {
        fetchErrors.push(jsonUrl + ' (' + response.status + ')');
        continue;
      }

      return await response.json();
    } catch (error) {
      fetchErrors.push(jsonUrl + ' (' + ((error && error.message) || 'fetch failed') + ')');
    }
  }

  throw new Error('Could not fetch OpenAPI JSON. Tried: ' + fetchErrors.join('; '));
}

function UMLGeneratorButtonRuntime(props) {
  var minZoom = props && typeof props.minZoom === 'number' ? props.minZoom : 0.35;
  var maxZoom = props && typeof props.maxZoom === 'number' ? props.maxZoom : 4;
  var initialZoom = props && typeof props.initialZoom === 'number' ? props.initialZoom : 1;
  var api = props && props.api;
  var schema = props && props.schema;
  var openApiUrl = props && (props.openApiUrl || props.sourceUrl || props.url);
  var parsedSchema = umlTryParseSelector(schema);
  var schemaTitle = parsedSchema ? parsedSchema.alias : '';
  var configurationWarnings = umlConfigurationWarnings(api, schema);
  var openLabel = (props && props.openLabel) || (schemaTitle ? 'Generate schema diagram for ' + schemaTitle : 'Generate schema diagram');
  var loadingLabel = schemaTitle ? 'Generating schema diagram for ' + schemaTitle + '…' : 'Generating schema diagram…';
  var title = (props && props.title) || (schemaTitle ? schemaTitle + ' schema diagram' : 'Schema diagram');
  var themeModeState = useState(umlDetectReadMeThemeMode(props));
  var themeMode = themeModeState[0];
  var setThemeMode = themeModeState[1];
  var umlTheme = umlThemeFromMode(themeMode, props);

  var overlayRef = useRef(null);
  var viewerRef = useRef(null);
  var triggerRef = useRef(null);
  var closeButtonRef = useRef(null);
  var dragStartRef = useRef({ pointerX: 0, pointerY: 0, x: 0, y: 0 });

  var svgState = useState('');
  var svg = svgState[0];
  var setSvg = svgState[1];

  var svgThemeModeState = useState('');
  var svgThemeMode = svgThemeModeState[0];
  var setSvgThemeMode = svgThemeModeState[1];

  var openState = useState(false);
  var isOpen = openState[0];
  var setIsOpen = openState[1];

  var loadingState = useState(false);
  var isLoading = loadingState[0];
  var setIsLoading = loadingState[1];

  var errorState = useState('');
  var error = errorState[0];
  var setError = errorState[1];

  var zoomState = useState(umlClamp(initialZoom, minZoom, maxZoom));
  var zoom = zoomState[0];
  var setZoom = zoomState[1];

  var panState = useState({ x: 0, y: 0 });
  var pan = panState[0];
  var setPan = panState[1];

  var draggingState = useState(false);
  var isDragging = draggingState[0];
  var setIsDragging = draggingState[1];

  var closeModal = useCallback(function () {
    setIsOpen(false);
    setIsDragging(false);
  }, []);

  var resetView = useCallback(
    function () {
      setZoom(umlClamp(initialZoom, minZoom, maxZoom));
      setPan({ x: 0, y: 0 });
    },
    [initialZoom, minZoom, maxZoom]
  );

  var generateAndOpen = useCallback(
    async function () {
      setIsOpen(true);
      setError('');
      resetView();

      if (svg && svgThemeMode === themeMode) return;

      setIsLoading(true);
      try {
        if (!api) throw new Error('Add an api prop, for example api="PIS".');
        if (!schema) throw new Error('Add a schema prop, for example schema="OBWriteDomesticConsent4.properties.Data.properties.Initiation:OBDomestic2".');

        var openApiDocument = await umlFetchOpenApiDocument(api, {
          openApiUrl: openApiUrl,
        });
        setSvg(generateUmlSvg(openApiDocument, schema, umlTheme));
        setSvgThemeMode(themeMode);
      } catch (generationError) {
        setError((generationError && generationError.message) || 'Failed to generate the schema diagram.');
      } finally {
        setIsLoading(false);
      }
    },
    [api, schema, svg, svgThemeMode, themeMode, umlTheme, resetView, openApiUrl]
  );

  var zoomIn = useCallback(
    function () {
      setZoom(function (value) {
        return umlClamp(Number((value + 0.2).toFixed(2)), minZoom, maxZoom);
      });
    },
    [minZoom, maxZoom]
  );

  var zoomOut = useCallback(
    function () {
      setZoom(function (value) {
        return umlClamp(Number((value - 0.2).toFixed(2)), minZoom, maxZoom);
      });
    },
    [minZoom, maxZoom]
  );

  var downloadSvg = useCallback(
    function () {
      if (!svg || typeof document === 'undefined') return;

      var blob = new Blob([svg], { type: 'image/svg+xml;charset=utf-8' });
      var url = URL.createObjectURL(blob);
      var link = document.createElement('a');
      link.href = url;
      link.download = umlDownloadFilename(schema) + '.svg';
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      URL.revokeObjectURL(url);
    },
    [svg, schema]
  );

  var onOverlayMouseDown = useCallback(
    function (event) {
      if (event.target === overlayRef.current) closeModal();
    },
    [closeModal]
  );

  var onWheel = useCallback(
    function (event) {
      event.preventDefault();

      var viewer = viewerRef.current;
      if (!viewer) return;

      var rect = viewer.getBoundingClientRect();
      var pointX = event.clientX - rect.left;
      var pointY = event.clientY - rect.top;
      var delta = event.deltaY > 0 ? -0.15 : 0.15;
      var nextZoom = umlClamp(Number((zoom + delta).toFixed(2)), minZoom, maxZoom);
      if (nextZoom === zoom) return;

      var scaleRatio = nextZoom / zoom;
      setZoom(nextZoom);
      setPan({
        x: pointX - (pointX - pan.x) * scaleRatio,
        y: pointY - (pointY - pan.y) * scaleRatio,
      });
    },
    [zoom, pan.x, pan.y, minZoom, maxZoom]
  );

  var onPointerDown = useCallback(
    function (event) {
      event.preventDefault();
      setIsDragging(true);
      dragStartRef.current = {
        pointerX: event.clientX,
        pointerY: event.clientY,
        x: pan.x,
        y: pan.y,
      };
    },
    [pan.x, pan.y]
  );

  var onPointerMove = useCallback(
    function (event) {
      if (!isDragging) return;
      setPan({
        x: dragStartRef.current.x + event.clientX - dragStartRef.current.pointerX,
        y: dragStartRef.current.y + event.clientY - dragStartRef.current.pointerY,
      });
    },
    [isDragging]
  );

  var onPointerUp = useCallback(function () {
    setIsDragging(false);
  }, []);

  useEffect(
    function () {
      if (!isOpen) {
        if (triggerRef.current && triggerRef.current.focus) triggerRef.current.focus();
        return undefined;
      }

      var previousOverflow = document.body.style.overflow;
      document.body.style.overflow = 'hidden';

      if (closeButtonRef.current && closeButtonRef.current.focus) closeButtonRef.current.focus();

      function onKeyDown(event) {
        if (event.key === 'Escape') {
          event.preventDefault();
          closeModal();
          return;
        }
        if (event.key === '+' || event.key === '=') {
          event.preventDefault();
          zoomIn();
          return;
        }
        if (event.key === '-') {
          event.preventDefault();
          zoomOut();
          return;
        }
        if (event.key === '0') {
          event.preventDefault();
          resetView();
        }
      }

      window.addEventListener('keydown', onKeyDown);

      return function () {
        document.body.style.overflow = previousOverflow;
        window.removeEventListener('keydown', onKeyDown);
      };
    },
    [isOpen, closeModal, zoomIn, zoomOut, resetView]
  );

  useEffect(
    function () {
      function syncThemeMode() {
        var nextThemeMode = umlDetectReadMeThemeMode(props);
        setThemeMode(function (currentThemeMode) {
          return currentThemeMode === nextThemeMode ? currentThemeMode : nextThemeMode;
        });
      }

      syncThemeMode();

      var observers = [];
      if (typeof MutationObserver !== 'undefined') {
        umlAvailableDocuments().forEach(function (doc) {
          [doc.documentElement, doc.body].filter(Boolean).forEach(function (element) {
            var observer = new MutationObserver(syncThemeMode);
            observer.observe(element, {
              attributes: true,
              attributeFilter: [
                'class',
                'style',
                'data-theme',
                'data-color-mode',
                'data-color-scheme',
                'data-readme-color-scheme',
                'color-scheme',
              ],
            });
            observers.push(observer);
          });
        });
      }

      var mediaQuery = null;
      try {
        mediaQuery =
          typeof window !== 'undefined' && window.matchMedia
            ? window.matchMedia('(prefers-color-scheme: dark)')
            : null;
        if (mediaQuery && mediaQuery.addEventListener) mediaQuery.addEventListener('change', syncThemeMode);
        else if (mediaQuery && mediaQuery.addListener) mediaQuery.addListener(syncThemeMode);
      } catch (error) {
        mediaQuery = null;
      }

      return function () {
        observers.forEach(function (observer) {
          observer.disconnect();
        });
        if (mediaQuery && mediaQuery.removeEventListener) mediaQuery.removeEventListener('change', syncThemeMode);
        else if (mediaQuery && mediaQuery.removeListener) mediaQuery.removeListener(syncThemeMode);
      };
    },
    [props && props.theme]
  );

  return (
    <div className="my-4">
      {configurationWarnings.length ? (
        <div
          role="alert"
          style={{
            display: 'inline-block',
            maxWidth: '720px',
            border: '1px solid ' + umlTheme.warningBorder,
            borderRadius: '8px',
            background: umlTheme.warningBackground,
            color: umlTheme.warning,
            padding: '10px 12px',
            fontSize: '13px',
            fontWeight: 600,
            lineHeight: 1.45,
          }}
        >
          <strong>Schema diagram is not configured.</strong>
          <ul style={{ margin: '6px 0 0 18px', padding: 0 }}>
            {configurationWarnings.map(function (warning) {
              return <li key={warning}>{warning}</li>;
            })}
          </ul>
          <pre
            style={{
              margin: '10px 0 0',
              border: '1px solid ' + umlTheme.warningBorder,
              borderRadius: '6px',
              background: umlTheme.modalBackground,
              color: umlTheme.text,
              padding: '8px',
              overflowX: 'auto',
              fontSize: '12px',
              fontWeight: 500,
            }}
          >
            {'<UMLGeneratorButton api="PIS" schema="OBWriteDomesticConsent4" />'}
          </pre>
          <p style={{ margin: '8px 0 0' }}>
            See the{' '}
            <a
              href="https://github.com/OpenBankingUK/ReadMe-Custom-Components/tree/main/uml-generator#schema-diagram-generator-for-readme"
              target="_blank"
              rel="noreferrer"
              style={{ color: 'inherit', textDecoration: 'underline' }}
            >
              UML generator README
            </a>{' '}
            for supported APIs and schema selector examples.
          </p>
        </div>
      ) : (
        <button
          ref={triggerRef}
          type="button"
          onClick={generateAndOpen}
          disabled={isLoading}
          style={{
            display: 'inline-flex',
            alignItems: 'center',
            gap: '8px',
            border: '1px solid ' + umlTheme.primaryBorder,
            borderRadius: '8px',
            background: umlTheme.primary,
            color: umlTheme.primaryText,
            padding: '9px 14px',
            fontSize: '13px',
            fontWeight: 700,
            cursor: isLoading ? 'not-allowed' : 'pointer',
            opacity: isLoading ? 0.7 : 1,
          }}
        >
          {isLoading ? loadingLabel : openLabel}
        </button>
      )}

      {isOpen ? (
        <div
          ref={overlayRef}
          role="dialog"
          aria-modal="true"
          aria-label={title + ' viewer'}
          onMouseDown={onOverlayMouseDown}
          style={{
            position: 'fixed',
            zIndex: 2147483647,
            inset: 0,
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            padding: '12px',
            background: 'rgba(0, 0, 0, 0.75)',
          }}
        >
          <div
            style={{
              position: 'relative',
              display: 'flex',
              flexDirection: 'column',
              width: '100%',
              maxWidth: '1280px',
              height: '90vh',
              background: umlTheme.modalBackground,
              color: umlTheme.text,
              borderRadius: '12px',
              boxShadow: '0 20px 50px rgba(0, 0, 0, 0.35)',
              overflow: 'hidden',
            }}
          >
            <div
              style={{
                display: 'flex',
                alignItems: 'center',
                justifyContent: 'space-between',
                gap: '8px',
                borderBottom: '1px solid ' + umlTheme.modalBorder,
                padding: '10px 12px',
              }}
            >
              <p style={{ margin: 0, color: umlTheme.text, fontSize: '14px', fontWeight: 700 }}>{title}</p>
              <div style={{ display: 'flex', alignItems: 'center', gap: '6px', flexWrap: 'wrap' }}>
                <button type="button" onClick={zoomOut} disabled={!svg}>
                  -
                </button>
                <button type="button" onClick={zoomIn} disabled={!svg}>
                  +
                </button>
                <button type="button" onClick={resetView} disabled={!svg}>
                  Reset
                </button>
                <button type="button" onClick={downloadSvg} disabled={!svg}>
                  Download SVG
                </button>
                <button ref={closeButtonRef} type="button" onClick={closeModal}>
                  Close
                </button>
              </div>
            </div>

            <div
              ref={viewerRef}
              onWheel={svg ? onWheel : undefined}
              onPointerMove={onPointerMove}
              onPointerUp={onPointerUp}
              onPointerLeave={onPointerUp}
              style={{
                position: 'relative',
                flex: 1,
                overflow: 'hidden',
                background: umlTheme.viewerBackground,
              }}
            >
              {isLoading ? (
                <div
                  style={{
                    position: 'absolute',
                    inset: 0,
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    color: umlTheme.primaryBorder,
                    fontSize: '14px',
                    fontWeight: 700,
                  }}
                >
                  {loadingLabel}
                </div>
              ) : null}

              {error ? (
                <div
                  style={{
                    position: 'absolute',
                    inset: 0,
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    padding: '18px',
                    color: umlTheme.error,
                    fontSize: '14px',
                    textAlign: 'center',
                  }}
                >
                  {error}
                </div>
              ) : null}

              {svg && !error ? (
                <img
                  src={umlSvgToDataUrl(svg)}
                  alt={title}
                  draggable={false}
                  onPointerDown={onPointerDown}
                  style={{
                    position: 'absolute',
                    top: 0,
                    left: 0,
                    maxWidth: 'none',
                    userSelect: 'none',
                    cursor: isDragging ? 'grabbing' : 'grab',
                    transform: 'translate(' + pan.x + 'px, ' + pan.y + 'px) scale(' + zoom + ')',
                    transformOrigin: 'top left',
                  }}
                />
              ) : null}
            </div>
          </div>
        </div>
      ) : null}
    </div>
  );
}

  return <UMLGeneratorButtonRuntime {...props} />;
};
